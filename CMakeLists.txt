if(TARGET cliptiq_menu_native)
  if(APPLE)
    set_source_files_properties(platform/MenuHost.cpp PROPERTIES LANGUAGE OBJCXX)
    set(menu_backend cliptiq_macos)
  elseif(WIN32)
    set(menu_backend cliptiq_windows)
  else()
    set(menu_backend cliptiq_linux)
  endif()
  function(cliptiq_menu_host name mode)
    add_executable(${name} platform/MenuHost.cpp)
    target_compile_definitions(${name} PRIVATE CLIPTIQ_MENU_MODE=${mode})
    target_link_libraries(${name} PRIVATE ${menu_backend})
    if(mode EQUAL 2)
      target_link_libraries(${name} PRIVATE cliptiq_menu_native)
    endif()
    if(APPLE)
      target_compile_options(${name} PRIVATE -fobjc-arc)
      target_link_options(${name} PRIVATE -Wl,-dead_strip)
    elseif(MSVC)
      target_compile_options(${name} PRIVATE /utf-8 /Gy /Gw)
      target_link_options(${name} PRIVATE /OPT:REF /OPT:ICF)
    else()
      target_compile_options(${name} PRIVATE -ffunction-sections -fdata-sections)
      target_link_options(${name} PRIVATE -Wl,--gc-sections $<$<CONFIG:MinSizeRel>:-s>)
    endif()
  endfunction()
  cliptiq_menu_host(cliptiq_menu_host 2)
endif()
