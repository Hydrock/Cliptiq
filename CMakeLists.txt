cmake_minimum_required(VERSION 3.23)
project(Cliptiq VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CLIPTIQ_SANITIZE "Enable address/undefined sanitizers (Clang/GCC)" OFF)
option(CLIPTIQ_BUILD_SIZE_PROBE "Build linked core footprint fixture" OFF)
include(cmake/Dependencies.cmake)
include(CTest)
add_library(cliptiq_core STATIC core/src/storage.cpp core/src/history.cpp core/src/sha256.cpp core/src/policy.cpp)
target_include_directories(cliptiq_core PUBLIC core/include PRIVATE core/src)
target_link_libraries(cliptiq_core PRIVATE cliptiq_sqlite)
if(MSVC)
  target_compile_options(cliptiq_core PRIVATE /W4 /utf-8 /Gy /Gw $<$<CONFIG:MinSizeRel>:/O1>)
else()
  target_compile_options(cliptiq_core PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
endif()
if(CLIPTIQ_BUILD_SIZE_PROBE)
  add_executable(core_probe tests/size/core_probe.cpp)
  add_executable(core_policy_probe tests/size/core_probe.cpp)
  target_compile_definitions(core_policy_probe PRIVATE CLIPTIQ_WITH_POLICY=1)
  target_link_libraries(core_policy_probe PRIVATE cliptiq_core)
  target_link_libraries(core_probe PRIVATE cliptiq_core)
  if(MSVC)
    target_link_options(core_probe PRIVATE /OPT:REF /OPT:ICF)
    target_link_options(core_policy_probe PRIVATE /OPT:REF /OPT:ICF)
  elseif(APPLE)
    target_link_options(core_probe PRIVATE -Wl,-dead_strip)
    target_link_options(core_policy_probe PRIVATE -Wl,-dead_strip)
  else()
    target_link_options(core_probe PRIVATE -Wl,--gc-sections -s)
    target_link_options(core_policy_probe PRIVATE -Wl,--gc-sections -s)
  endif()
endif()
if(CLIPTIQ_SANITIZE AND NOT MSVC)
  target_compile_options(cliptiq_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cliptiq_core PUBLIC -fsanitize=address,undefined)
  target_compile_options(cliptiq_sqlite PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(BUILD_TESTING)
  add_executable(policy_test tests/unit/policy_test.cpp)
  target_link_libraries(policy_test PRIVATE cliptiq_core cliptiq_sqlite)
  if(MSVC)
    target_compile_options(policy_test PRIVATE /utf-8)
  endif()
  add_test(NAME policy COMMAND policy_test)
  set_tests_properties(policy PROPERTIES TIMEOUT 60)
  add_executable(history_test tests/unit/history_test.cpp)
  target_include_directories(history_test PRIVATE core/src)
  target_link_libraries(history_test PRIVATE cliptiq_core cliptiq_sqlite)
  if(MSVC)
    target_compile_options(history_test PRIVATE /utf-8)
  endif()
  add_test(NAME history COMMAND history_test)
  set_tests_properties(history PROPERTIES TIMEOUT 60)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME history_crash COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/history_crash.py $<TARGET_FILE:history_test>)
  set_tests_properties(history_crash PROPERTIES TIMEOUT 30)
endif()
