add_library(cliptiq_migration STATIC ../../core/src/migration.cpp)
target_include_directories(cliptiq_migration PRIVATE ../../core/src)
target_link_libraries(cliptiq_migration PUBLIC cliptiq_core PRIVATE cliptiq_sqlite)
add_executable(cliptiq_import main.cpp)
set_target_properties(cliptiq_import PROPERTIES OUTPUT_NAME cliptiq-import)
target_link_libraries(cliptiq_import PRIVATE cliptiq_migration)
if(WIN32)
 target_link_libraries(cliptiq_migration PRIVATE advapi32)
 target_compile_options(cliptiq_migration PRIVATE /utf-8 /Gy /Gw)
else()
 target_compile_options(cliptiq_migration PRIVATE -ffunction-sections -fdata-sections)
endif()
if(APPLE)
 add_executable(cliptiq_import_macos main.mm)
 set_target_properties(cliptiq_import_macos PROPERTIES OUTPUT_NAME cliptiq-import-macos)
 target_compile_options(cliptiq_import_macos PRIVATE -fobjc-arc)
 target_link_libraries(cliptiq_import_macos PRIVATE cliptiq_migration "-framework Foundation")
endif()
if(WIN32)
 target_link_options(cliptiq_import PRIVATE /OPT:REF /OPT:ICF)
elseif(APPLE)
 target_link_options(cliptiq_import PRIVATE -Wl,-dead_strip)
 target_link_options(cliptiq_import_macos PRIVATE -Wl,-dead_strip)
else()
 target_link_options(cliptiq_import PRIVATE -Wl,--gc-sections $<$<CONFIG:MinSizeRel>:-s>)
endif()
