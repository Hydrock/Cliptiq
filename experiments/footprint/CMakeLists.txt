cmake_minimum_required(VERSION 3.22)
project(CliptiqFootprint LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
option(PROBE_FTS5 "Include SQLite FTS5" OFF)
option(PROBE_STATIC_CRT "Windows static CRT" ON)
option(PROBE_GTK4 "Linux GTK4 instead of GTK3" OFF)
option(PROBE_OCR "Measure one native/external OCR fixture operation" OFF)
if(NOT EXISTS "${SQLITE_SOURCE}/sqlite3.c")
  message(FATAL_ERROR "Run python tools/fetch-footprint-deps.py; set SQLITE_SOURCE")
endif()
if(MSVC)
  if(PROBE_STATIC_CRT)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
    set(RUST_CRT -Ctarget-feature=+crt-static)
  endif()
  add_compile_options(/O1 /utf-8 /Gy /Gw)
  add_link_options(/OPT:REF /OPT:ICF /INCREMENTAL:NO /LTCG)
else()
  add_compile_options(-Os -ffunction-sections -fdata-sections)
  if(APPLE)
    add_link_options(-Wl,-dead_strip)
  else()
    add_link_options(-Wl,--gc-sections -s)
  endif()
endif()
add_library(probe_sqlite STATIC "${SQLITE_SOURCE}/sqlite3.c")
target_include_directories(probe_sqlite PUBLIC "${SQLITE_SOURCE}")
target_compile_definitions(probe_sqlite PRIVATE SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
if(PROBE_FTS5)
  target_compile_definitions(probe_sqlite PRIVATE SQLITE_ENABLE_FTS5)
endif()
find_program(RUSTC rustc REQUIRED)
set(RUST_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}probe_rust${CMAKE_STATIC_LIBRARY_SUFFIX}")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
  COMMAND "${RUSTC}" --edition=2021 --crate-type=staticlib --crate-name=cliptiq_probe
    -Copt-level=z -Cpanic=abort -Clto=thin -Ccodegen-units=1 ${RUST_CRT}
    "${CMAKE_CURRENT_SOURCE_DIR}/native-rust/core.rs" -o "${RUST_LIBRARY}"
  DEPENDS native-rust/core.rs VERBATIM)
add_custom_target(build_rust DEPENDS "${RUST_LIBRARY}")
add_library(rust_core STATIC IMPORTED GLOBAL)
set_target_properties(rust_core PROPERTIES IMPORTED_LOCATION "${RUST_LIBRARY}")
add_dependencies(rust_core build_rust)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE)
  set(HOST native_windows.cpp)
  set(OS_LIBS user32 shell32 psapi advapi32 userenv ws2_32 bcrypt ntdll)
  if(PROBE_OCR)
    set(OCR_SOURCE ocr_windows.cpp)
    list(APPEND OS_LIBS windowsapp gdi32)
  endif()
elseif(APPLE)
  enable_language(OBJCXX)
  set(HOST native_macos.mm)
  set(OS_LIBS "-framework AppKit" "-framework Security" "-framework SystemConfiguration" c++ resolv)
  if(PROBE_OCR)
    set(OCR_SOURCE ocr_macos.mm)
    list(APPEND OS_LIBS "-framework Vision")
  endif()
else()
  find_package(PkgConfig REQUIRED)
  if(PROBE_GTK4)
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4)
  else()
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
  endif()
  set(HOST native_linux.cpp)
  set(OS_LIBS PkgConfig::GTK dl pthread m)
  if(PROBE_OCR)
    set(OCR_SOURCE ocr_linux.cpp)
  endif()
endif()
if(PROBE_OCR)
  add_compile_definitions(PROBE_OCR)
endif()
foreach(LANGUAGE cpp rust)
  set(CORE native-cpp/core.cpp)
  if(LANGUAGE STREQUAL rust)
    set(CORE "")
  endif()
  add_executable(probe_${LANGUAGE} main.cpp fixture.cpp ${HOST} ${CORE} ${OCR_SOURCE})
  if(WIN32 AND PROBE_OCR)
    # Recent C++/WinRT uses standard coroutines rather than MSVC's removed
    # experimental coroutine header. Other baseline lanes remain C++17.
    set_property(TARGET probe_${LANGUAGE} PROPERTY CXX_STANDARD 20)
  endif()
  if(LANGUAGE STREQUAL rust)
    target_link_libraries(probe_${LANGUAGE} PRIVATE rust_core)
  endif()
  target_link_libraries(probe_${LANGUAGE} PRIVATE probe_sqlite ${OS_LIBS})
  add_test(NAME fixture_${LANGUAGE} COMMAND probe_${LANGUAGE} --self-test "${CMAKE_CURRENT_BINARY_DIR}/${LANGUAGE}-snippets.xml")
  add_test(NAME lifecycle_${LANGUAGE} COMMAND probe_${LANGUAGE} --lifecycle-smoke)
  set_tests_properties(lifecycle_${LANGUAGE} PROPERTIES TIMEOUT 8)
endforeach()
