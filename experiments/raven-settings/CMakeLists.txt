# Opt-in spike. Main application and its native settings implementation are unchanged.
set(CLIPTIQ_RAVEN_DIR "${PROJECT_SOURCE_DIR}/.toolchains/raven" CACHE PATH "Pinned Raven checkout")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CLIPTIQ_RAVEN_DIR}"
  OUTPUT_VARIABLE raven_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT raven_revision STREQUAL "9f0cfa6fe3da33022a577e06a6ea843a5ef3f76e")
  message(FATAL_ERROR "Unexpected Raven revision: ${raven_revision}")
endif()
set(RAVEN_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_WEB OFF CACHE BOOL "" FORCE)
set(RAVEN_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory("${CLIPTIQ_RAVEN_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/raven" EXCLUDE_FROM_ALL)
foreach(variant native raven)
  add_executable(settings_${variant} WIN32 main.cpp)
  target_link_libraries(settings_${variant} PRIVATE cliptiq_core user32 comctl32)
  target_compile_options(settings_${variant} PRIVATE /utf-8 /W4 /Gy /Gw /O1)
  target_link_options(settings_${variant} PRIVATE /OPT:REF /OPT:ICF)
  target_compile_definitions(settings_${variant} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  set_target_properties(settings_${variant} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/settings-bin")
  if(variant STREQUAL "native")
    target_sources(settings_native PRIVATE "${PROJECT_SOURCE_DIR}/platform/windows/Settings.cpp")
  else()
    target_compile_definitions(settings_raven PRIVATE SETTINGS_RAVEN=1)
    target_link_libraries(settings_raven PRIVATE raven::raven)
  endif()
endforeach()
